class Solution {
    public boolean halvesAreAlike(String s) {
        String vowels = "aeiouAEIOU";
        int mid = s.length() / 2;
        int diff = 0;

        for (int i = 0; i < mid; i++) {
            if (vowels.indexOf(s.charAt(i)) != -1) diff++;
            if (vowels.indexOf(s.charAt(mid + i)) != -1) diff--;
        }

        return diff == 0;
    }
}
